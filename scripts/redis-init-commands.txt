# ============================================================================
# FlashDrop Redis Initialization Commands
# Run these commands to initialize Redis with sample data
# ============================================================================
#
# ARCHITECTURE NOTE:
# Redis is the SINGLE SOURCE OF TRUTH for all ephemeral/real-time data:
#   - Inventory counters (flashdrop:inventory:*)
#   - Cart sessions (flashdrop:cart:*)
#   - Active reservations (flashdrop:reservation:*)
#   - Flash sale counters (flashdrop:flashsale:*)
#   - Rate limiting (flashdrop:ratelimit:*)
#
# PostgreSQL stores ONLY persistent/completed data:
#   - Completed orders, payment transactions, user accounts
#   - Inventory audit log (changes AFTER checkout completes)
#
# ============================================================================

# Connect to Redis:
# redis-cli -h flashdrop-dev-redis.yhhwe0.0001.use1.cache.amazonaws.com -p 6379

# ============================================================================
# FLASH SALE DATA
# Real-time counters and configuration for active flash sales
# ============================================================================

# Flash sale sold counters (real-time tracking)
SET flashdrop:flashsale:1:sold 75
SET flashdrop:flashsale:3:sold 40

# Flash sale configuration (cached from PostgreSQL flash_sales table)
HSET flashdrop:flashsale:1 id "flash-1" product_id "1" original_price "170" sale_price "129" max_quantity "100" is_active "1"
HSET flashdrop:flashsale:3 id "flash-3" product_id "3" original_price "220" sale_price "179" max_quantity "50" is_active "1"

# ============================================================================
# CART SESSIONS (Example - created dynamically by cart-service)
# TTL: 1800 seconds (30 minutes)
# ============================================================================
# Example cart creation (DO NOT RUN - just for reference):
# HSET flashdrop:cart:session123 id "cart-uuid" session_id "session123" items "[]" created_at "2026-01-19T12:00:00Z"
# EXPIRE flashdrop:cart:session123 1800

# ============================================================================
# RESERVATIONS (Example - created dynamically during checkout)
# TTL: 600 seconds (10 minutes)
# These are the SINGLE SOURCE OF TRUTH for active inventory holds
# PostgreSQL does NOT track active reservations
# ============================================================================
# Example reservation (DO NOT RUN - just for reference):
# SET flashdrop:reservation:session123:1:9 1
# EXPIRE flashdrop:reservation:session123:1:9 600
#
# When checkout completes:
#   1. DEL flashdrop:reservation:session123:1:9
#   2. PostgreSQL: INSERT INTO inventory_audit_log (change_type='sale', ...)
#
# When reservation expires (user abandons checkout):
#   1. Redis TTL auto-deletes the key
#   2. Background job: INCR flashdrop:inventory:1:9 (restore stock)

# ============================================================================
# RATE LIMITING (Example - created dynamically by API)
# TTL: 60 seconds
# ============================================================================
# Example rate limit (DO NOT RUN - just for reference):
# INCR flashdrop:ratelimit:192.168.1.1
# EXPIRE flashdrop:ratelimit:192.168.1.1 60

# ============================================================================
# VERIFICATION COMMANDS
# ============================================================================

# Count all FlashDrop keys
KEYS flashdrop:*

# Check inventory for a specific product/size
GET flashdrop:inventory:1:9

# Check flash sale status
HGETALL flashdrop:flashsale:1
GET flashdrop:flashsale:1:sold

# Check total inventory for Product 1 (all sizes)
# MGET flashdrop:inventory:1:7 flashdrop:inventory:1:7.5 flashdrop:inventory:1:8 ...
